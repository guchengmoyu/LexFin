# 07 · 前端设计

> 技术栈：**Vue 3.4 + Vite 5 + TypeScript + Element Plus + Pinia + Vue Router 4 + Axios**。
> 这套组合是国内企业级中后台最经典的方案，组件库文档齐全、招聘与交接成本低。

## 7. SSE 流式接收实现（★ 关键代码）

浏览器的 `EventSource` **只支持 GET**，而本系统问答接口是 POST，
因此必须用 `fetch` + `ReadableStream` 手动解析 SSE。

```ts
// src/utils/sse.ts
export interface SseHandlers {
  onMeta?: (d: MetaEvent) => void
  onToken?: (delta: string) => void
  onDone?: (d: DoneEvent) => void
  onError?: (e: { code: number; message: string }) => void
}

export async function postSse(
  url: string,
  body: unknown,
  handlers: SseHandlers,
  signal?: AbortSignal,
): Promise<void> {
  const res = await fetch(url, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Accept: 'text/event-stream',
      Authorization: `Bearer ${localStorage.getItem('finrag_token') ?? ''}`,
    },
    body: JSON.stringify(body),
    signal,
  })

  if (!res.ok || !res.body) {
    const err = await res.json().catch(() => ({ code: res.status, message: '网络异常' }))
    handlers.onError?.(err)
    return
  }

  const reader = res.body.getReader()
  const decoder = new TextDecoder('utf-8')
  let buffer = ''

  const dispatch = (raw: string) => {
    let event = 'message'
    const dataLines: string[] = []
    for (const line of raw.split('\n')) {
      if (line.startsWith('event:')) event = line.slice(6).trim()
      else if (line.startsWith('data:')) dataLines.push(line.slice(5).trim())
    }
    if (!dataLines.length) return
    let payload: any
    try { payload = JSON.parse(dataLines.join('\n')) } catch { return }
    if (event === 'meta') handlers.onMeta?.(payload)
    else if (event === 'token') handlers.onToken?.(payload.delta ?? '')
    else if (event === 'done') handlers.onDone?.(payload)
    else if (event === 'error') handlers.onError?.(payload)
    // event === 'ping' 忽略
  }

  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })
    const parts = buffer.split(/\r?\n\r?\n/)     // SSE 事件以空行分隔
    buffer = parts.pop() ?? ''                    // 最后一段可能不完整，留到下一轮
    for (const p of parts) if (p.trim()) dispatch(p)
  }
  if (buffer.trim()) dispatch(buffer)             // 冲刷末尾残留
}
```

**在 `chat.ts` 中的使用**

```ts
async function send(question: string, opts: AskOptions) {
  const ctrl = new AbortController()
  abortCtrl.value = ctrl
  streaming.value = true

  messages.value.push({ role: 'user', content: question, status: 'done' })
  const assistant: ChatMessage = { role: 'assistant', content: '', status: 'streaming' }
  messages.value.push(assistant)
  citations.value = []

  try {
    await postSse(`${API_BASE}/qa/ask/stream`, { question, session_no: sessionNo.value, ...opts }, {
      onMeta: (d) => {
        sessionNo.value = d.session_no
        assistant.messageId = d.message_id
        citations.value = d.citations ?? []
      },
      onToken: (delta) => { assistant.content += delta },
      onDone: (d) => {
        assistant.status = 'done'
        assistant.answerType = d.answer_type
        assistant.confidence = d.confidence
        assistant.confidenceLevel = d.confidence_level
        assistant.latencyMs = d.latency_ms
        assistant.suggestions = d.suggestions
      },
      onError: (e) => {
        assistant.status = 'error'
        assistant.error = e.message
        ElMessage.error(e.message)
      },
    }, ctrl.signal)
  } catch (e: any) {
    if (e?.name === 'AbortError') assistant.status = 'done'   // 用户主动停止
    else { assistant.status = 'error'; assistant.error = '连接中断' }
  } finally {
    streaming.value = false
    abortCtrl.value = null
  }
}
```

**必须注意的三点**

1. **Nginx 必须关闭缓冲**：`proxy_buffering off;` 与 `X-Accel-Buffering: no`，
   否则 SSE 会被攒批下发，流式效果完全失效（这是最常见的踩坑点）。
2. **Vue 响应式更新**：直接改 `assistant.content += delta` 对 `ref` 数组内对象有效，
   但**不能**用 `messages.value[i] = {...}` 重建对象（会丢失流式期间的引用，导致闪烁）。
3. **刷新丢失**：流式过程中刷新页面会丢失当前回答，服务端已落库，
   刷新后从 `GET /qa/sessions/{session_no}` 可完整回看（含引用）。

## 8. API 封装与拦截器

```ts
// src/api/request.ts
const service = axios.create({ baseURL: import.meta.env.VITE_API_BASE_URL, timeout: 30000 })

service.interceptors.request.use((cfg) => {
  const token = localStorage.getItem('finrag_token')
  if (token) cfg.headers.Authorization = `Bearer ${token}`
  cfg.headers['X-Request-ID'] = crypto.randomUUID().replace(/-/g, '').slice(0, 16)
  return cfg
})

service.interceptors.response.use(
  (res) => {
    const body = res.data
    if (body.code === 0) return body.data          // 直接解包 data，业务层拿到的就是数据
    ElMessage.error(body.message || '请求失败')
    return Promise.reject(body)
  },
  (err) => {
    const body = err.response?.data
    const code = body?.code ?? 50000
    if ([40100, 40101, 40102].includes(code)) {
      useUserStore().logout()
      router.replace({ path: '/login', query: { redirect: router.currentRoute.value.fullPath } })
      ElMessage.warning('登录已失效，请重新登录')
    } else if (code === 42900) {
      ElMessage.warning(body.message || '操作过于频繁，请稍后再试')
    } else {
      ElMessage.error(body?.message || '服务异常，请稍后重试')
    }
    return Promise.reject(body ?? err)
  },
)
```

**注意**：拦截器直接 `return body.data` 意味着 `res.data.data` 才是业务数据；
上传等需要 `multipart` 的接口单独用 `headers: { 'Content-Type': 'multipart/form-data' }`。

## 9. 权限控制

```ts
// 1. 路由守卫
router.beforeEach(async (to) => {
  const user = useUserStore()
  if (to.meta.public) return true
  if (!user.isLogin) return { path: '/login', query: { redirect: to.fullPath } }
  if (!user.info) await user.fetchMe()          // 首次进入拉取用户信息
  if (to.meta.roles && !user.hasRole(...(to.meta.roles as string[]))) {
    ElMessage.warning('无访问权限'); return { path: '/chat' }
  }
  return true
})

// 2. 路由 meta 声明
{ path: '/documents', component: DocumentListView, meta: { roles: ['editor', 'admin'] } }

// 3. 按钮级指令
app.directive('role', {
  mounted(el, binding) {
    const user = useUserStore()
    if (!user.hasRole(...[].concat(binding.value))) el.remove()
  },
})
// 用法：<el-button v-role="['editor','admin']">上传文档</el-button>
```

```nginx
server {
    listen 80;
    server_name _;
    root /usr/share/nginx/html;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;      # history 路由回退
    }

    location /api/ {
        proxy_pass http://backend:8000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        # ★ SSE 必需：关闭缓冲，否则流式失效
        proxy_buffering off;
        proxy_cache off;
        proxy_set_header Connection '';
        chunked_transfer_encoding on;

        # 大文件上传与长回答

        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
    }

    gzip on;
    gzip_types text/plain text/css application/javascript application/json;
```
